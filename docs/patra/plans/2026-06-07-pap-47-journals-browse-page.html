    @Builder
public record VenueBrowseFilter(
    String keyword, VenueBrowseSort sort,
    List<String> subjects, List<String> jcrQuartiles,
    List<String> casQuartiles, Boolean casTop,
    Boolean isOpenAccess, Boolean doaj, List<String> countryCodes) {
  public VenueBrowseFilter {
    if (sort == null) sort = VenueBrowseSort.IMPACT_FACTOR;
    subjects = subjects != null ? List.copyOf(subjects) : List.of();
    jcrQuartiles = jcrQuartiles != null ? List.copyOf(jcrQuartiles) : List.of();
    casQuartiles = casQuartiles != null ? List.copyOf(casQuartiles) : List.of();
    countryCodes = countryCodes != null ? List.copyOf(countryCodes) : List.of();
  }
}
  4. 步骤 4：创建 VenueBrowseFacets 读模型

    record，5+ 字段用 @Builder。分组维度为 List<FacetCount>，布尔维度为 long 计数。内嵌 FacetCount(String value, long count)（参数 2 个 → 静态 of()）。集合字段紧凑构造器防御性拷贝。

    @Builder
public record VenueBrowseFacets(
    List<FacetCount> subjects, List<FacetCount> jcrQuartiles,
    List<FacetCount> casQuartiles, List<FacetCount> countries,
    long casTop, long openAccess, long doaj) {
  public VenueBrowseFacets {
    subjects = subjects != null ? List.copyOf(subjects) : List.of();
    jcrQuartiles = jcrQuartiles != null ? List.copyOf(jcrQuartiles) : List.of();
    casQuartiles = casQuartiles != null ? List.copyOf(casQuartiles) : List.of();
    countries = countries != null ? List.copyOf(countries) : List.of();
  }
  public record FacetCount(String value, long count) {
    public static FacetCount of(String value, long count) { return new FacetCount(value, count); }
  }
}

    -- base（page 与 facet 共用；实现者抽成可复用片段或 CTE）
SELECT v.id, v.title AS name, v.abbreviated_title AS abbr, v.country_code,
       v.cited_by_count, v.image_object_key AS cover_object_key, v.issn_l,
       (v.open_access->>'isOa')::boolean      AS is_open_access,
       (v.open_access->>'isInDoaj')::boolean  AS is_in_doaj,
       jcr.impact_factor, jcr.jif_quartile AS jcr_quartile, jcr.subject AS jcr_subject,
       cas.major_category AS cas_major_category, cas.major_quartile AS cas_major_quartile,
       cas.is_top_journal AS cas_is_top, v.id AS founded_year_src  -- founded_year 来源同现有实现
FROM cat_venue v
LEFT JOIN LATERAL (SELECT * FROM cat_venue_jcr_rating r WHERE r.venue_id=v.id ORDER BY r.year DESC LIMIT 1) jcr ON true
LEFT JOIN LATERAL (SELECT * FROM cat_venue_cas_rating r WHERE r.venue_id=v.id ORDER BY r.year DESC, r.edition DESC LIMIT 1) cas ON true
WHERE v.deleted_at IS NULL

    page 过滤条件（追加到 base 外层或同层 WHERE）：keyword 前缀 LIKE（保留现有 ESCAPE '!'）、各多值 = ANY、casTop/isOpenAccess/doaj 布尔（(:flag IS NULL OR col = :flag)）。ORDER BY 沿用现有 sort switch。注意：foundedYear 字段保持现有取法（如来自 publication_profile JSONB 或某列）——以现有 VenueBrowseRow 投影为准，不要臆造。

  3. 步骤 3：VenueDao 加 facet 聚合查询（GREEN）

    每个分组维度一个原生 @Query，复用 base，drill-down = 应用「keyword + 除本维度外所有过滤」后 GROUP BY。例（jcr 维度，省略本维度自身的 jcr 过滤）：

    SELECT b.jcr_quartile AS value, count(*) AS count FROM ( <base> ) b
WHERE (:keyword IS NULL OR b.name ILIKE :keyword ESCAPE '!' OR b.abbr ILIKE :keyword ESCAPE '!')
  AND (:subjects   IS NULL OR b.jcr_subject       = ANY(:subjects))
  AND (:casQuartiles IS NULL OR b.cas_major_quartile = ANY(:casQuartiles))
  AND (:countryCodes IS NULL OR b.country_code     = ANY(:countryCodes))
  AND (:casTop IS NULL OR b.cas_is_top = :casTop)
  AND (:isOpenAccess IS NULL OR b.is_open_access = :isOpenAccess)
  AND (:doaj IS NULL OR b.is_in_doaj = :doaj)
  AND b.jcr_quartile IS NOT NULL
GROUP BY b.jcr_quartile

    subjects/casQuartiles/countries 维度同构（各自省略本维度的过滤、对本维度列 GROUP BY、IS NOT NULL）。3 个布尔维度用 SELECT count(*)（应用「其它所有筛选 + 本布尔=true」）。实现提示：base 片段重复，建议抽成 SQL 常量字符串或 DB 视图；7 条查询可接受（记 tradeoff：未来可合并 GROUPING SETS）。
