import cn.hutool.core.lang.Assert;

/// 摘要段落值对象——主摘要 / 翻译摘要 / Entity 序列化 / 读模型共享的唯一 section 定义。
///
/// 存储形态为有序数组 `[{"label","text"}]`：保 section 顺序、容纳重复 label 与无 label 段。
///
/// @param label 段落标签（如 BACKGROUND / METHODS；无标签段为 null，空白串归一化为 null）
/// @param text 段落内容（非空）
/// @author linqibin
/// @since 0.6.0
public record PublicationAbstractSection(String label, String text) implements Serializable {

  @Serial private static final long serialVersionUID = 1L;

  /// 紧凑构造器：label 空白归一化为 null 并 trim，text 非空校验。
  public PublicationAbstractSection {
    label = (label == null || label.isBlank()) ? null : label.trim();
    Assert.notBlank(text, "摘要段落内容不能为空");
  }

  /// 创建摘要段落。
  ///
  /// @param label 段落标签（可为 null / 空白）
  /// @param text 段落内容
  /// @return 段落值对象
  public static PublicationAbstractSection of(String label, String text) {
    return new PublicationAbstractSection(label, text);
  }

  /// 是否带标签段落（STRUCTURED 判型依据）。
  ///
  /// @return true 如果 label 非空
  public boolean isLabeled() {
    return label != null;
  }
}

  /// 紧凑构造器：校验 + 防御性拷贝。
  public PublicationAuthorSnapshot {
    Assert.isTrue(order >= 1, "作者顺序必须 >= 1");

  /// 派生展示名：集体名优先；否则 "LastName ForeName"，ForeName 缺则用 Initials，
  /// 仅 LastName 则单独成名；全空返回 null（调用方据此跳过该作者）；超长按 200 截断。
  ///
  /// @param lastName 姓
  /// @param foreName 名
  /// @param initials 缩写
  /// @param collectiveName 集体作者名
  /// @return 展示名，或 null（无可用姓名）
  public static String deriveDisplayName(
      String lastName, String foreName, String initials, String collectiveName) {
    String collective = StrUtil.trimToNull(collectiveName);
    if (collective != null) {
      return truncate(collective);
    }
    String last = StrUtil.trimToNull(lastName);
    if (last == null) {
      return null;
    }
    String fore = StrUtil.trimToNull(foreName);
    if (fore != null) {
      return truncate(last + " " + fore);
    }
    String init = StrUtil.trimToNull(initials);
    if (init != null) {
      return truncate(last + " " + init);
    }
    return truncate(last);
  }

  /// 按展示名列宽截断。
  private static String truncate(String value) {
    return value.length() <= DISPLAY_NAME_MAX_LENGTH
        ? value
        : value.substring(0, DISPLAY_NAME_MAX_LENGTH);
  }

  public boolean isCollective() {
    return collectiveName != null;
  }

      /// 构建摘要信息（两形态一等公民）。
  ///
  /// 判型：≥1 个带 Label 段 → STRUCTURED（含混合，无 Label 段以 label=null 保位）；
  /// 全部无 Label → UNSTRUCTURED（sections 置空，文本合并落 plainText）。
  /// plainText 统一在此生成："LABEL: text" / 无 label 直接 text，段间 \n 连接。
  ///
  /// @param publication 规范化文献
  /// @return 摘要值对象，无摘要返回 null
  private PublicationAbstract buildPublicationAbstract(CanonicalPublication publication) {
    Abstract abstractContent = publication.getAbstractContent();
    if (abstractContent == null) {
      return null;
    }

    String copyright = abstractContent.getCopyright();
    List<AbstractSection> rawSections =
        abstractContent.getSections() != null ? abstractContent.getSections() : List.of();
    List<PublicationAbstractSection> sections =
        rawSections.stream()
            .filter(s -> s.getContent() != null && !s.getContent().isBlank())
            .map(s -> PublicationAbstractSection.of(s.getLabel(), s.getContent()))
            .toList();

    if (sections.isEmpty()) {
      // 源侧无段落：退回 Canonical 层的 text（API 路径等其他来源可能只给 text）
      String plainText = abstractContent.getText();
      if (plainText != null && !plainText.isBlank()) {
        return PublicationAbstract.ofPlainText(plainText, copyright);
      }
      return null;
    }

    String plainText = joinSectionsAsPlainText(sections);
    boolean structured = sections.stream().anyMatch(PublicationAbstractSection::isLabeled);
    if (structured) {
      return PublicationAbstract.ofBoth(plainText, sections, copyright);
    }
    // 全部无 Label：非结构化，sections 不入库（Entity 层置 null）
    return PublicationAbstract.ofPlainText(plainText, copyright);
  }

  /// 段落拼接为纯文本（读端 plainText 的唯一生成点）。
  private String joinSectionsAsPlainText(List<PublicationAbstractSection> sections) {
    return sections.stream()
        .map(s -> s.isLabeled() ? s.label() + ": " + s.text() : s.text())
        .collect(Collectors.joining("\n"));
  }

  /// @param publication 规范化文献
  /// @return 作者快照列表（可能为空，不为 null）
  private List<PublicationAuthorSnapshot> buildAuthorSnapshots(CanonicalPublication publication) {
    List<Author> authors = publication.getAuthors();
    if (authors == null || authors.isEmpty()) {
      return List.of();
    }

    List<PublicationAuthorSnapshot> snapshots = new ArrayList<>();
    int order = 1;
    for (Author author : authors) {
      if (author == null) {
        continue;
      }
      String lastName = StrUtil.trimToNull(author.getLastName());
      String foreName = StrUtil.trimToNull(author.getForeName());
      String initials = StrUtil.trimToNull(author.getInitials());
      String suffix = StrUtil.trimToNull(author.getSuffix());
      String collectiveName = StrUtil.trimToNull(author.getOrganizationName());
      String displayName =
          PublicationAuthorSnapshot.deriveDisplayName(lastName, foreName, initials, collectiveName);
      if (displayName == null) {
        log.debug("跳过无有效姓名的作者：importBatch={}", importBatch);
        continue;
      }

      List<PublicationAuthorAffiliationSnapshot> affiliations =
          buildAffiliationSnapshots(author);

      snapshots.add(
          PublicationAuthorSnapshot.builder()
              .order(order)
              .lastName(lastName)
              .foreName(foreName)
              .initials(initials)
              .suffix(suffix)
              .collectiveName(collectiveName)
              .displayName(displayName)
              .orcid(normalizeOrcid(extractOrcid(author)))
              .firstAuthor(order == 1)
              .equalContribution(Boolean.TRUE.equals(author.getEqualContribution()))
              .affiliations(affiliations)
              .build());
      order++;
    }
    return snapshots;
  }

  /// 构建机构快照（保 XML 顺序，1 起连续编号）。
  private List<PublicationAuthorAffiliationSnapshot> buildAffiliationSnapshots(Author author) {
    if (author.getAffiliations() == null || author.getAffiliations().isEmpty()) {
      return List.of();
    }
    List<PublicationAuthorAffiliationSnapshot> result = new ArrayList<>();
    int affOrder = 1;
    for (Affiliation affiliation : author.getAffiliations()) {
      String name = affiliation != null ? StrUtil.trimToNull(affiliation.getName()) : null;
      if (name == null) {
        continue;
      }
      result.add(PublicationAuthorAffiliationSnapshot.of(affOrder++, name));
    }
    return result;
  }

  /// 提取作者 ORCID 原始值（identifiers 中 type=ORCID 的首个）。
  private String extractOrcid(Author author) {
    if (author.getIdentifiers() == null) {
      return null;
    }
    return author.getIdentifiers().stream()
        .filter(id -> id.getType() == PublicationIdentifierType.ORCID)
        .map(CanonicalPublication.Identifier::getValue)
        .filter(v -> v != null && !v.isBlank())
        .findFirst()
        .orElse(null);
  }

  /// ORCID 归一化：去 URL 前缀 → 大写 → 格式 + ISO 7064 校验位校验，失败返回 null。
  private String normalizeOrcid(String raw) {
    if (raw == null || raw.isBlank()) {
      return null;
    }
    String value = raw.trim();
    value = StrUtil.removePrefixIgnoreCase(value, "https://orcid.org/");
    value = StrUtil.removePrefixIgnoreCase(value, "http://orcid.org/");
    value = value.toUpperCase(Locale.ROOT);
    if (!Orcid.isValid(value)) {
      return null;
    }
    Orcid orcid = Orcid.of(value);
    return orcid.isChecksumValid() ? orcid.value() : null;
  }

  private PublicationAggregate loadAuthors(PublicationAggregate aggregate) {
    Long id = aggregate.getId() != null ? aggregate.getId().value() : null;
    if (id == null) {
      return aggregate;
    }
    List<PublicationAuthorEntity> rows =
        publicationAuthorDao.findAllByPublicationIdOrderByAuthorOrderAsc(id);
    if (rows.isEmpty()) {
      return aggregate;
    }
    Map<Long, List<PublicationAuthorAffiliationEntity>> affByPubAuthor =
        publicationAuthorAffiliationDao
            .findAllByPubAuthorIdIn(rows.stream().map(PublicationAuthorEntity::getId).toList())
            .stream()
            .collect(Collectors.groupingBy(PublicationAuthorAffiliationEntity::getPubAuthorId));
    aggregate.attachAuthors(
        rows.stream()
            .map(e -> jpaConverter.toAuthorSnapshot(
                e, affByPubAuthor.getOrDefault(e.getId(), List.of())))
            .toList());
    return aggregate;
  }

import gzip, sys
import xml.etree.ElementTree as ET

def text_of(el):
    return "".join(el.itertext()).strip() if el is not None else ""

def main(xml_gz: str, pmids_file: str) -> None:
    imported = set(line.strip() for line in open(pmids_file) if line.strip())
    with_abstract = valid_authors = with_orcid = skipped_authors = 0
    with gzip.open(xml_gz) as f:
        for _, art in ET.iterparse(f, events=("end",)):
            if art.tag != "PubmedArticle":
                continue
